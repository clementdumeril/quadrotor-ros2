cmake_minimum_required(VERSION 3.8)
project(quad_control C CXX)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:-Wall> $<$<COMPILE_LANGUAGE:CXX>:-Wextra> $<$<COMPILE_LANGUAGE:CXX>:-Wpedantic>)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(quad_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)

add_executable(position_controller src/position_controller.cpp)
target_include_directories(position_controller PRIVATE include ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(position_controller rclcpp nav_msgs geometry_msgs std_srvs quad_interfaces)
install(TARGETS position_controller DESTINATION lib/${PROJECT_NAME})

# NMPC: the solver C code is generated by nmpc/generate_solver.py and committed; building it
# needs only the acados libraries (ACADOS_SOURCE_DIR, default ~/acados).
set(ACADOS_DIR "$ENV{ACADOS_SOURCE_DIR}")
if(NOT ACADOS_DIR)
  set(ACADOS_DIR "$ENV{HOME}/acados")
endif()
if(EXISTS "${ACADOS_DIR}/lib/libacados.so")
  set(GEN ${CMAKE_CURRENT_SOURCE_DIR}/nmpc/c_generated_code)
  add_library(quadrotor_nmpc_solver STATIC
    ${GEN}/acados_solver_quadrotor.c
    ${GEN}/quadrotor_model/quadrotor_expl_ode_fun.c
    ${GEN}/quadrotor_model/quadrotor_expl_vde_forw.c
    ${GEN}/quadrotor_model/quadrotor_expl_vde_adj.c)
  set_target_properties(quadrotor_nmpc_solver PROPERTIES C_STANDARD 99 POSITION_INDEPENDENT_CODE ON)
  target_include_directories(quadrotor_nmpc_solver PUBLIC
    ${GEN} ${ACADOS_DIR}/include ${ACADOS_DIR}/include/acados
    ${ACADOS_DIR}/include/blasfeo/include ${ACADOS_DIR}/include/hpipm/include)
  target_link_libraries(quadrotor_nmpc_solver PUBLIC
    ${ACADOS_DIR}/lib/libacados.so ${ACADOS_DIR}/lib/libhpipm.so ${ACADOS_DIR}/lib/libblasfeo.so m)

  add_executable(nmpc_controller src/nmpc_controller.cpp)
  target_include_directories(nmpc_controller PRIVATE include ${EIGEN3_INCLUDE_DIRS})
  target_link_libraries(nmpc_controller quadrotor_nmpc_solver)
  ament_target_dependencies(nmpc_controller rclcpp nav_msgs std_msgs std_srvs quad_interfaces)
  # RPATH (not RUNPATH) so that libacados also finds libhpipm and libblasfeo at run time.
  set_target_properties(nmpc_controller PROPERTIES INSTALL_RPATH "${ACADOS_DIR}/lib")
  target_link_options(nmpc_controller PRIVATE -Wl,--disable-new-dtags)
  install(TARGETS nmpc_controller DESTINATION lib/${PROJECT_NAME})
else()
  message(WARNING "acados not found in ${ACADOS_DIR}: nmpc_controller is not built")
endif()

install(DIRECTORY config DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_reference test/test_reference.cpp)
  target_include_directories(test_reference PRIVATE include ${EIGEN3_INCLUDE_DIRS})
  ament_target_dependencies(test_reference rclcpp quad_interfaces)
endif()

ament_package()
